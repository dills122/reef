        private fun canonical(n: JsonNode): String {
            val output = java.io.StringWriter()
            mapper.factory.createGenerator(output).use { generator ->
                fun write(value: JsonNode) {
                    when {
                        value.isObject -> {
                            generator.writeStartObject()
                            value.properties().sortedBy { it.key }.forEach { (key, child) ->
                                generator.writeFieldName(key); write(child)
                            }
                            generator.writeEndObject()
                        }
                        value.isArray -> {
                            generator.writeStartArray(); value.forEach { write(it) }; generator.writeEndArray()
                        }
                        value.isTextual -> generator.writeString(value.textValue())
                        value.isIntegralNumber -> generator.writeNumber(value.bigIntegerValue())
                        value.isBoolean -> generator.writeBoolean(value.booleanValue())
                        value.isNull -> generator.writeNull()
                        else -> mapper.writeTree(generator, value)
                    }
                }
                write(n)
            }
            return output.toString()
        }
